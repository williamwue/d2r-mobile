cmake_minimum_required(VERSION 3.22.1)
project(d2rm_probe LANGUAGES CXX ASM)
add_library(d2rm_probe SHARED probe.cpp bridge_probe.cpp)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_sources(d2rm_probe PRIVATE bridge_arm64.S)
endif()
target_compile_features(d2rm_probe PRIVATE cxx_std_17)
target_compile_options(d2rm_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(d2rm_probe PRIVATE dl)
target_link_options(d2rm_probe PRIVATE "-Wl,-z,max-page-size=16384")

add_library(d2rm_graphics_probe SHARED present_probe.cpp)
target_compile_features(d2rm_graphics_probe PRIVATE cxx_std_17)
target_compile_options(d2rm_graphics_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(d2rm_graphics_probe PRIVATE vulkan android)
target_link_options(d2rm_graphics_probe PRIVATE "-Wl,-z,max-page-size=16384")
